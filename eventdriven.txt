@app.websocket("/ws/metrics/dashboard")
async def websocket_dashboard(websocket: WebSocket):
    await websocket.accept()
    try:
        dashboard_request = await websocket.receive_json()
        interval = dashboard_request.get("interval", 10)
        panels = dashboard_request.get("panels", [])

        # 🧠 Store previous values to detect changes
        previous_results = {}

        while True:
            full_response = {
                "timestamp": int(asyncio.get_event_loop().time()),
                "results": []
            }

            for panel in panels:
                panel_id = panel.get("panel_id")
                panel_result = {"panel_id": panel_id, "queries": []}

                for idx, query_def in enumerate(panel.get("queries", [])):
                    query_key = f"{panel_id}_{idx}"
                    try:
                        promql = build_advanced_promql(
                            query_def.get("metric"),
                            query_def.get("org"),
                            query_def.get("region"),
                            query_def.get("instance"),
                            query_def.get("name_regex"),
                            query_def.get("job"),
                            query_def.get("use_rate", False),
                            query_def.get("window", "5m"),
                            query_def.get("agg", "sum"),
                            query_def.get("group_by"),
                            query_def.get("multiply")
                        )

                        result = query(promql)
                        new_data = result.get("data", {})

                        # ✅ Check if result changed
                        if previous_results.get(query_key) != new_data:
                            previous_results[query_key] = new_data
                            panel_result["queries"].append({
                                "promql": promql,
                                "data": new_data
                            })

                    except Exception as e:
                        panel_result["queries"].append({
                            "promql": None,
                            "error": str(e)
                        })

                # 🔄 Only include this panel if it has changed queries
                if panel_result["queries"]:
                    full_response["results"].append(panel_result)

            # 🔄 Only send if there are any changes
            if full_response["results"]:
                await websocket.send_text(json.dumps(full_response))

            await asyncio.sleep(interval)

    except WebSocketDisconnect:
        print("Dashboard WebSocket disconnected.")
    except Exception as e:
        await websocket.send_text(json.dumps({"error": str(e)}))
